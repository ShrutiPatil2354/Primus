cmake_minimum_required(VERSION 3.20)
project(primus_reference_core LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(mujoco CONFIG QUIET)
if(TARGET mujoco::mujoco)
    add_executable(primus_reflex reflex.cpp)
    target_link_libraries(primus_reflex PRIVATE mujoco::mujoco)
    add_executable(primus_spine spine.cpp)
    target_link_libraries(primus_spine PRIVATE mujoco::mujoco)
else()
    message(STATUS "MuJoCo CMake package not found; reflex/spine targets are disabled")
endif()

if(UNIX AND NOT APPLE)
    find_package(OpenCV QUIET COMPONENTS core imgproc videoio)
    if(OpenCV_FOUND)
        add_executable(primus_cam cam.cpp)
        target_include_directories(primus_cam PRIVATE ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(primus_cam PRIVATE ${OpenCV_LIBS})
    else()
        message(STATUS "OpenCV development package not found; shared-memory camera target is disabled")
    endif()
else()
    message(STATUS "primus_cam uses POSIX shared memory and V4L2; target is Linux-only")
endif()
